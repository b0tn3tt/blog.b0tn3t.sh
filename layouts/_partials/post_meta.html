{{- /* Project override of PaperMod's post_meta. Date + reading time are shown in the page's own language: Chinese on the zh-cn version, English on RU/EN (RU keeps an English UI by request). Go-native .Date.Format is used so the month/format is a literal we control, not Hugo's locale-based time.Format. */ -}} {{- $meta := slice }} {{- $datefmt := cond (eq .Lang "zh-cn") "2006年1月2日" "January 2, 2006" }} {{- if not .Date.IsZero -}} {{- $meta = $meta | append (printf "%s" (.Date.Format "2006-01-02T15:04:05-07:00") (.Date.Format $datefmt)) }} {{- end }} {{- if (.Param "ShowReadingTime") -}} {{- $meta = $meta | append (printf "%s" (i18n "read_time" .ReadingTime | default (printf "%d min" .ReadingTime))) }} {{- end }} {{- if (.Param "ShowWordCount") -}} {{- $meta = $meta | append (printf "%s" (i18n "words" .WordCount | default (printf "%d words" .WordCount))) }} {{- end }} {{- if not (.Param "hideAuthor") -}} {{- with (partial "author.html" .) }} {{- $meta = $meta | append (printf "%s" .) }} {{- end }} {{- end }} {{/* Combine all meta information into a single string with separators and render it as HTML.*/}} {{- with $meta }} {{- delimit . " · " | safeHTML -}} {{- end -}}
